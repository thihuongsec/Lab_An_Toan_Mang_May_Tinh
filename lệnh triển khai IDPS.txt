#Tải các gói hỗ trợ snort
sudo apt-get install -y build-essential
sudo apt-get install -y libpcap-dev libpcre3-dev libdumbnet-dev 
sudo apt-get install -y bison flex 
sudo apt-get install -y zlib1g-dev liblzma-dev openssl libssl-dev 
Tạo thư mục để tí tải Snort về sẽ lưu vào đó
mkdir ~/snort_src 
cd ~/snort_src 

#Tải 2 gói gồm daq và snort, dưới đây là tải daq
sudo wget https://www.snort.org/downloads/snort/daq-2.0.7.tar.gz 
sudo tar -xvzf daq-2.0.7.tar.gz 
cd daq-2.0.7
sudo ./configure 
sudo make
sudo make install

#quay lại tải gói snort tiếp
cd ..
sudo wget https://snort.org/downloads/snort/snort-2.9.20.tar.gz 
sudo tar -zxf snort-2.9.20.tar.gz 
sudo ./configure  --enable-sourcefire --disable-open-appid 
sudo make
sudo make install

#lệnh giúp cập nhật các thay đổi trong thư mục, file mà không cần reboot
sudo ldconfig 

*\tạo một link sẽ cho phép ta khởi động snort từ bất kì vị trí nào
trong terminal mà không cần truy cập vào path đang chứa snort\*
sudo ln /usr/local/bin/snort /usr/sbin/snort

#tạo các thư mục cho snort_src
sudo mkdir /etc/snort 
sudo mkdir /etc/snort/rules 
sudo mkdir /etc/snort/rules/iplists 
sudo mkdir /etc/snort/preproc_rules 
sudo mkdir /usr/local/lib/snort_dynamicrules 
sudo mkdir /etc/snort/so_rules 

#tạo các file chứa luật cơ bản của snort 
sudo touch /etc/snort/rules/iplists/black_list.rules 
sudo touch /etc/snort/rules/iplists/white_list.rules 
sudo touch /etc/snort/rules/local.rules 
sudo touch /etc/snort/sid-msg.map 
 
#tạo thư mục chứa các log cho snort
sudo mkdir /var/log/snort 
sudo mkdir /var/log/snort/archived_logs 

#tạo bản sao cho các file configure của snort
cd snort_src/snort-2.9.20/etc
sudo cp *.conf* /etc/snort  
sudo cp *.map /etc/snort 
sudo cp *.dtd /etc/snort 
cd ~/snort_src/snort-2.9.20/src/dynamic-preprocessors/build/usr/local/lib/snort_dynamicpreprocessor/ 
sudo cp * /usr/local/lib/snort_dynamicpreprocessor/ 

#thư mục chứa ip:  /etc/snort/rules/iplists 

#chỉnh sửa các tham số trong tệp tin /etc/snort/snort.conf theo y như sau
sudo nano /etc/snort/snort.conf 

#kiểm tra hoạt động và lưu cấu hình của snort bằng 1 trong 2 lệnh sau
sudo snort -i ens33 -c /etc/snort/snort.conf -T

#để xem và theo dõi hoạt động dò quét của snort trong thời gian thực
sudo tail -f /var/log/snort/alert
